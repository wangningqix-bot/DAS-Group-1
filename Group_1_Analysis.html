1.3 Dataset Overview

The dataset contains 11 original variables:

Variable Description
Total.Household.Income Annual household income (in Philippine Pesos)
Region All households are from CAR region
Total.Food.Expenditure Annual food expenditure (in Philippine Pesos)
Household.Head.Sex Male/Female
Household.Head.Age Age in years
Type.of.Household Single Family / Extended Family / Non-related
Total.Number.of.Family.members Household size
House.Floor.Area Square meters
House.Age Age of the house in years
Number.of.bedrooms Count of bedrooms
Electricity Whether household has electricity access (1=Yes, 0=No)

2.1 Data Loading and Preprocessing

# Read the data 
data <- read.csv("dataset01.csv")  
# Rename variables for easier handling 
df <- data %>%   
  rename(     
   income = Total.Household.Income,     
   region = Region,     
   food_exp = Total.Food.Expenditure,     
   head_sex = Household.Head.Sex,    
   head_age = Household.Head.Age,     
   hh_type = Type.of.Household,    
   family_size = Total.Number.of.Family.members,     
   floor_area = House.Floor.Area,     
   house_age = House.Age,     
   bedrooms = Number.of.bedrooms,     
   electricity = Electricity   
   ) %>%   
   mutate(     
   head_sex = as.factor(head_sex),     
   hh_type = as.factor(hh_type),     
   electricity = as.factor(electricity),     
   income_log = log(income),     
   food_exp_log = log(food_exp),     
   floor_area_log = log(floor_area),     
   food_ratio = food_exp / income * 100,     
   age_group = cut(head_age,                      
                   breaks = c(0, 30, 40, 50, 60, 70, 100),                                      labels = c("<30", "30-39", "40-49", "50-59", "60-69", "70+")),     
    income_group = cut(income,                         
                       breaks = quantile(income, probs = c(0, 0.25, 0.5, 0.75, 1)),                        
                       labels = c("Low", "Lower-Middle", "Upper-Middle", "High")),     
                       
    bedrooms_group = cut(bedrooms,                           
                         breaks = c(-1, 0, 1, 2, 3, 10),              
                         labels = c("0", "1", "2", "3", "4+"))   
)
